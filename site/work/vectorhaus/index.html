02 / Developers

Bring a function. We’ll bring the GPUs.

    Wrap a function

    Run vh deploy

    We build, snapshot and warm it, usually in under two minutes.

  3. 03

    Call it from anywhere

    A streaming HTTPS endpoint with keys, retries and logs built in.

app.py

import vectorhaus as vh

app = vh.App(
    "voice-agent",
    gpu="L40S",
    min_replicas=0,    # scale to zero when idle
    max_replicas=64,
)

@app.endpoint(stream=True)
def reply(audio: bytes):
    text = app.model("asr-small").transcribe(audio)
    # tokens stream back as soon as they exist
    for token in app.model("agent-8b").generate(text):
        yield token

client.ts

import { Vectorhaus } from "@vectorhaus/client";

const vh = new Vectorhaus({ apiKey: process.env.VH_API_KEY });

// stream tokens back while the user is still talking
const stream = await vh.call("voice-agent/reply", {
  input: audioChunk,
  stream: true,
});

}
